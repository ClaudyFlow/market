### 3.1 【区块链风格购物记录】

**创新等级：★★★★★**

#### 3.1.1 设计背景

传统电商平台的订单记录只是简单的数据库存储，用户无法直观感受到交易的"可信度"和"不可篡改性"。我们将区块链的思想融入消费记录，让每一笔交易都有类似"区块"的验证机制。

**哈希生成算法**：

```typescript
// 改进的哈希算法（类似区块链SHA-256简化版）
const generateHash = (data: string): string => {
  let hash = 0
  const salt = 'MARKET_BLOCKCHAIN_2024'

  // 混合盐值增强安全性
  const saltedData = salt + data + reverseString(salt)

  for (let i = 0; i < saltedData.length; i++) {
    const char = saltedData.charCodeAt(i)
    // 位运算混淆
    hash = ((hash << 5) - hash) + char
    hash = hash & hash // 强制转换为32位整数
  }

  // 十六进制表示
  let hexHash = Math.abs(hash).toString(16)

  // 补齐8位
  while (hexHash.length < 8) {
    hexHash = '0' + hexHash
  }

  return hexHash.toUpperCase()
}

// 字符串反转辅助函数
const reverseString = (str: string): string => {
  return str.split('').reverse().join('')
}

// 生成完整记录哈希
const generateRecordHash = (record: PurchaseRecord, previousHash: string): string => {
  const data = [
    record.status,
    record.timestamp,
    record.description,
    previousHash,
    record.orderId || '',
    record.operator || ''
  ].join('|')

  return generateHash(data)
}
```

```java
// 购物记录服务
@Service
public class PurchaseRecordService {

    @Autowired
    private PurchaseRecordRepository repository;

    // 创建购物记录（类似区块链的"挖矿"）
    @Transactional
    public PurchaseRecord createRecord(Long orderId, String status) {
        Order order = orderRepository.findById(orderId)
            .orElseThrow(() -> new BusinessException("订单不存在"));

        // 获取上一条记录
        String previousHash = "GENESIS"; // 创世区块
        PurchaseRecord lastRecord = repository.findTopByOrderIdOrderByCreatedAtDesc(orderId);
        if (lastRecord != null) {
            previousHash = lastRecord.getHash();
        }

        // 构建记录
        PurchaseRecord record = new PurchaseRecord();
        record.setOrderId(orderId);
        record.setStatus(status);
        record.setTimestamp(LocalDateTime.now());
        record.setPreviousHash(previousHash);

        // 生成哈希
        String hash = computeHash(record);
        record.setHash(hash);
        record.setVerified(true);

        return repository.save(record);
    }

    // 计算哈希
    private String computeHash(PurchaseRecord record) {
        String data = String.join("|",
            record.getStatus(),
            record.getTimestamp().toString(),
            record.getPreviousHash(),
            record.getOrderId().toString()
        );

        try {
            MessageDigest md = MessageDigest.getInstance("SHA-256");
            byte[] hash = md.digest(data.getBytes(StandardCharsets.UTF_8));
            return bytesToHex(hash).toUpperCase().substring(0, 8);
        } catch (NoSuchAlgorithmException e) {
            throw new RuntimeException(e);
        }
    }

    // 验证记录完整性
    public boolean verifyRecord(Long recordId) {
        PurchaseRecord record = repository.findById(recordId)
            .orElseThrow(() -> new BusinessException("记录不存在"));

        String computedHash = computeHash(record);
        return computedHash.equals(record.getHash());
    }
}
```

#### 3.2.3 成长值获取算法

```java
@Service
public class GrowthValueService {

    // 成长值获取配置
    private static final Map<String, Integer> GROWTH_CONFIG = Map.of(
        "CONSUMPTION", 10,    // 每消费1元 = 10成长值
        "REVIEW", 50,         // 每评价一次 = 50成长值
        "CHECKIN", 10,        // 每次签到 = 10成长值（×连续天数倍数）
        "LIKE_RECEIVED", 5,   // 被点赞一次 = 5成长值
        "SHARE", 20           // 分享一次 = 20成长值
    );

    // 连续签到加成系数
    private static final Map<Integer, BigDecimal> CHECKIN_MULTIPLIER = Map.of(
        3, new BigDecimal("1.2"),   // 3天  +20%
        7, new BigDecimal("1.5"),   // 7天  +50%
        14, new BigDecimal("2.0"),  // 14天 +100%
        30, new BigDecimal("3.0")   // 30天 +200%
    );

    /**
     * 计算成长值
     */
    public int calculateGrowthValue(String action, BigDecimal amount, User user) {
        int baseValue = GROWTH_CONFIG.getOrDefault(action, 0);

        // 特殊处理：消费金额
        if ("CONSUMPTION".equals(action) && amount != null) {
            baseValue = amount.multiply(new BigDecimal("10")).intValue();
        }

        // 特殊处理：连续签到加成
        if ("CHECKIN".equals(action)) {
            int consecutiveDays = user.getConsecutiveCheckinDays();
            BigDecimal multiplier = getCheckinMultiplier(consecutiveDays);
            baseValue = new BigDecimal(baseValue)
                .multiply(multiplier)
                .intValue();
        }

        // 每日上限检查
        int dailyLimit = 500; // 每日最多500成长值
        return Math.min(baseValue, dailyLimit);
    }

    // 获取签到加成系数
    private BigDecimal getCheckinMultiplier(int consecutiveDays) {
        BigDecimal multiplier = BigDecimal.ONE;

        if (consecutiveDays >= 30) {
            multiplier = new BigDecimal("3.0");
        } else if (consecutiveDays >= 14) {
            multiplier = new BigDecimal("2.0");
        } else if (consecutiveDays >= 7) {
            multiplier = new BigDecimal("1.5");
        } else if (consecutiveDays >= 3) {
            multiplier = new BigDecimal("1.2");
        }

        return multiplier;
    }

    /**
     * 计算VIP等级
     */
    public int calculateVipLevel(int totalGrowthValue) {
        if (totalGrowthValue >= 100000) return 5; // 黑金
        if (totalGrowthValue >= 50000) return 4;  // 钻石
        if (totalGrowthValue >= 20000) return 3;  // 铂金
        if (totalGrowthValue >= 5000) return 2;   // 黄金
        if (totalGrowthValue >= 1000) return 1;  // 白银
        return 0;                                 // 普通
    }
}
```

```java
@Service
public class VipBenefitService {

    @Autowired
    private VipLevelRepository vipLevelRepository;

    /**
     * 计算VIP折扣
     */
    public BigDecimal calculateDiscount(int vipLevel, BigDecimal orderAmount) {
        VipLevel vip = vipLevelRepository.findByLevel(vipLevel)
            .orElseThrow(() -> new BusinessException("VIP等级不存在"));

        BigDecimal discountRate = vip.getDiscountRate(); // 如 0.92 表示92折

        return orderAmount.multiply(discountRate);
    }

    /**
     * 检查VIP特权
     */
    public VipPrivilege checkPrivilege(int vipLevel, String privilegeType) {
        VipLevel vip = vipLevelRepository.findByLevel(vipLevel)
            .orElseThrow(() -> new BusinessException("VIP等级不存在"));

        switch (privilegeType) {
            case "FREE_SHIPPING":
                // 免邮次数
                int freeShippingCount = vip.getFreeShippingCount();
                return new VipPrivilege(true, freeShippingCount, "次/月");

            case "PRIORITY_DELIVERY":
                // 优先发货
                boolean hasPriority = vip.getGrowthValueRequired() >= 5000;
                return new VipPrivilege(hasPriority, 0, "");

            case "EXCLUSIVE_SERVICE":
                // 专属客服
                return new VipPrivilege(vip.getExclusiveService(), 0, "");

            case "REFUND_PRIORITY":
                // 优先退款
                return new VipPrivilege(vip.getRefundPriority(), 0, "");

            default:
                return new VipPrivilege(false, 0, "");
        }
    }

    /**
     * 获得VIP每日积分
     */
    public int getDailyCredit(int vipLevel) {
        VipLevel vip = vipLevelRepository.findByLevel(vipLevel)
            .orElseThrow(() -> new BusinessException("VIP等级不存在"));

        return vip.getDailyCredit();
    }
}
```

### 3.3 【智能抽奖系统】

**创新等级：★★★★☆**

#### 3.3.1 设计背景

传统抽奖系统存在以下问题：
1. 奖池一次性被掏空
2. 同一用户重复中奖
3. 概率不透明，可被人为操控

我们设计了一套**概率可控 + 库存动态调整 + 防重复机制**的智能抽奖系统。

#### 3.3.2 系统架构

```
│  │   用户请求   │────▶│  限流检查   │────▶│  抽奖算法   │             │
```

```java
@Service
public class LotteryService {

    @Autowired
    private RedisService redisService;

    @Autowired
    private LotteryPrizeRepository prizeRepository;

    @Autowired
    private LotteryRecordRepository recordRepository;

    // 奖品基础概率配置
    private static final Map<Integer, BigDecimal> BASE_PROBABILITY = Map.of(
        1, new BigDecimal("0.001"),   // 一等奖 0.1%
        2, new BigDecimal("0.005"),   // 二等奖 0.5%
        3, new BigDecimal("0.02"),    // 三等奖 2%
        4, new BigDecimal("0.05"),   // 四等奖 5%
        5, new BigDecimal("0.10")    // 参与奖 10%
    );

    // 每用户每日最大抽奖次数
    private static final int DAILY_MAX_DRAWS = 10;

    /**
     * 抽奖入口
     */
    @Transactional
    public DrawResult draw(Long userId, Long activityId) {
        // 1. 获取分布式锁（防刷）
        String lockKey = String.format("lottery:lock:%d:%d", userId, activityId);
        boolean locked = redisService.tryLock(lockKey, 30, TimeUnit.SECONDS);
        if (!locked) {
            throw new BusinessException("抽奖太频繁，请稍后重试");
        }

        try {
            // 2. 检查抽奖次数
            checkDailyLimit(userId, activityId);

            // 3. 执行抽奖
            LotteryPrize prize = executeDraw(activityId);

            // 4. 扣减库存
            if (prize != null) {
                deductStock(prize.getId());
            }

            // 5. 记录中奖
            saveDrawRecord(userId, activityId, prize);

            // 6. 返回结果
            return buildResult(prize);

        } finally {
            redisService.unlock(lockKey);
        }
    }

    /**
     * 执行抽奖（核心算法）
     */
    private LotteryPrize executeDraw(Long activityId) {
        // 获取所有奖品
        List<LotteryPrize> prizes = prizeRepository.findByActivityId(activityId);

        // 计算总概率
        BigDecimal totalProbability = prizes.stream()
            .map(LotteryPrize::getProbability)
            .reduce(BigDecimal.ZERO, BigDecimal::add);

        // 生成随机数 [0, 1)
        BigDecimal random = BigDecimal.valueOf(Math.random());

        // 累计概率判定
        BigDecimal cumulative = BigDecimal.ZERO;
        for (LotteryPrize prize : prizes) {
            // 动态调整概率（根据库存）
            BigDecimal adjustedProb = adjustProbabilityByStock(prize);

            cumulative = cumulative.add(adjustedProb);

            if (random.compareTo(cumulative) <= 0) {
                // 中奖！
                return prize;
            }
        }

        // 未中奖（参与奖）
        return null;
    }

    /**
     * 根据库存动态调整概率
     *
     * 公式：adjusted_prob = base_prob × (remaining_stock / initial_stock)
     *
     * 效果：
     * - 库存充足时：实际概率 ≈ 基础概率
     * - 库存不足时：实际概率降低，避免超发
     * - 库存为0时：该奖品概率降为0
     */
    private BigDecimal adjustProbabilityByStock(LotteryPrize prize) {
        BigDecimal baseProb = prize.getProbability();
        int remainingStock = prize.getRemainingStock();
        int initialStock = prize.getTotalStock();

        if (remainingStock <= 0) {
            return BigDecimal.ZERO;
        }

        // 库存系数（0.5 ~ 1.5）
        BigDecimal stockFactor = new BigDecimal(remainingStock)
            .divide(new BigDecimal(initialStock), 4, RoundingMode.HALF_UP);

        // 防止系数过小（至少保留50%概率）
        stockFactor = stockFactor.max(new BigDecimal("0.5"));

        // 防止系数过大（最多150%概率）
        stockFactor = stockFactor.min(new BigDecimal("1.5"));

        return baseProb.multiply(stockFactor);
    }

    /**
     * 检查每日抽奖次数限制
     */
    private void checkDailyLimit(Long userId, Long activityId) {
        String countKey = String.format("lottery:count:%d:%d:%s",
            userId, activityId, LocalDate.now());

        Integer count = redisService.get(countKey);
        if (count != null && count >= DAILY_MAX_DRAWS) {
            throw new BusinessException("今日抽奖次数已用完，明天再来吧！");
        }

        // 增加计数
        redisService.incr(countKey);
        // 设置过期时间（到明天凌晨）
        redisService.expire(countKey, getSecondsUntilTomorrow());
    }

    /**
     * 扣减库存（Redis原子操作）
     */
    private void deductStock(Long prizeId) {
        String stockKey = String.format("lottery:stock:%d", prizeId);

        Long remaining = redisService.decr(stockKey);
        if (remaining < 0) {
            // 库存不足，回滚
            redisService.incr(stockKey);
            throw new BusinessException("抱歉，该奖品已发完");
        }

        // 同步到数据库
        lotteryPrizeRepository.decrementStock(prizeId);
    }
}
```

#### 3.3.4 抽奖概率模型

```java
/**
 * 抽奖概率计算器
 *
 * 使用负二项分布模拟真实抽奖场景
 * 核心思想：让用户感觉"再抽一次就中了"，但实际概率不变
 */
public class LotteryProbabilityCalculator {

    /**
     * 计算期望抽中次数
     *
     * @param n 抽奖次数
     * @param p 单次中奖概率
     * @return 期望中奖次数
     */
    public static double expectedWins(int n, double p) {
        return n * p;
    }

    /**
     * 计算至少中奖一次的概率
     *
     * P(at least one win) = 1 - P(no wins) = 1 - (1-p)^n
     *
     * @param n 抽奖次数
     * @param p 单次中奖概率
     * @return 至少中奖一次的概率
     */
    public static double probabilityAtLeastOneWin(int n, double p) {
        return 1 - Math.pow(1 - p, n);
    }

    /**
     * 模拟：计算"保底"机制所需抽奖次数
     *
     * 场景：如果概率是1%，需要抽多少次才能有99%的把握中奖？
     * 公式：1 - (1-p)^n >= 0.99
     * 推导：n >= ln(0.01) / ln(1-p)
     *
     * @param p 单次中奖概率
     * @param confidence 期望概率（0-1）
     * @return 需要的抽奖次数
     */
    public static int calculateGuaranteedWins(double p, double confidence) {
        return (int) Math.ceil(Math.log(1 - confidence) / Math.log(1 - p));
    }

    /**
     * 输出概率表（用于配置参考）
     */
    public static void printProbabilityTable() {
        System.out.println("抽奖次数 | 单次概率 | 至少中1次概率 | 期望中奖次数");
        System.out.println("---------|---------|--------------|-------------");

        double p = 0.01; // 1%概率

        for (int n : List.of(1, 5, 10, 20, 50, 100, 200, 459)) {
            double atLeastOne = probabilityAtLeastOneWin(n, p);
            double expected = expectedWins(n, p);

            System.out.printf("%8d | %7.2f | %12.2f | %12.2f%n",
                n, p * 100, atLeastOne * 100, expected);
        }
    }

    public static void main(String[] args) {
        printProbabilityTable();

        // 计算1%概率，99%保底需要多少次
        int guaranteed = calculateGuaranteedWins(0.01, 0.99);
        System.out.println("\n1%概率，99%保底需要: " + guaranteed + " 次");
    }
}
```

```java
@Service
public class SensitiveWordFilter {

    // DFA状态机
    private Map<Character, Map<String, Object>> dfa = new HashMap<>();

    // 敏感词库
    private volatile Set<String> sensitiveWords = new HashSet<>();

    // 是否初始化
    private volatile boolean initialized = false;

    /**
     * 初始化DFA状态机
     * 时间复杂度：O(n) 其中n为敏感词总长度
     * 构建后查询复杂度：O(m) 其中m为待检测文本长度
     */
    public synchronized void buildDFA(List<String> words) {
        if (words == null || words.isEmpty()) {
            return;
        }

        dfa.clear();

        for (String word : words) {
            if (word == null || word.trim().isEmpty()) {
                continue;
            }

            Map<Character, Map<String, Object>> current = dfa;

            for (char c : word.toCharArray()) {
                // 不区分大小写
                c = Character.toLowerCase(c);

                current = current.computeIfAbsent(c, k -> {
                    Map<String, Object> newState = new HashMap<>();
                    newState.put("children", new HashMap<Character, Map<String, Object>>());
                    newState.put("isEnd", false);
                    return newState;
                });

                // 获取下一状态
                Map<String, Object> nextState = current.get("children");
                current = (Map<Character, Map<String, Object>>) nextState.get(c);
            }

            // 标记结束
            current.put("isEnd", true);
            current.put("word", word);
        }

        sensitiveWords.addAll(words);
        initialized = true;
    }

    /**
     * 检测敏感词
     * @param text 待检测文本
     * @return 第一个匹配到的敏感词，未找到返回null
     */
    public String findFirst(String text) {
        if (!initialized || text == null) {
            return null;
        }

        Map<Character, Map<String, Object>> current = dfa;

        for (int i = 0; i < text.length(); i++) {
            char c = Character.toLowerCase(text.charAt(i));

            Map<String, Object> state = current.get(c);

            if (state == null) {
                // 重新从根开始
                current = dfa;
                continue;
            }

            // 检查是否是结束状态
            if (Boolean.TRUE.equals(state.get("isEnd"))) {
                return (String) state.get("word");
            }

            // 转移到下一个状态
            current = (Map<Character, Map<String, Object>>) state.get("children");
        }

        return null;
    }

    /**
     * 替换敏感词
     * @param text 待处理文本
     * @param replacement 替换字符，如"*"
     * @return 替换后的文本
     */
    public String replace(String text, char replacement) {
        if (!initialized || text == null) {
            return text;
        }

        StringBuilder result = new StringBuilder();
        Map<Character, Map<String, Object>> current = dfa;
        int matchStart = -1;
        int matchEnd = -1;
        String matchedWord = null;

        for (int i = 0; i < text.length(); i++) {
            char c = Character.toLowerCase(text.charAt(i));

            Map<String, Object> state = current.get(c);

            if (state == null) {
                // 无匹配，重置
                if (matchStart >= 0) {
                    // 用替换字符填充
                    for (int j = matchStart; j < i; j++) {
                        result.append(replacement);
                    }
                    matchStart = -1;
                } else {
                    result.append(text.charAt(i));
                }
                current = dfa;
                continue;
            }

            // 记录匹配起始位置
            if (matchStart < 0) {
                matchStart = i;
            }

            // 检查是否是结束状态
            if (Boolean.TRUE.equals(state.get("isEnd"))) {
                matchEnd = i;
                matchedWord = (String) state.get("word");
            }

            // 转移到下一个状态
            current = (Map<Character, Map<String, Object>>) state.get("children");
        }

        // 处理末尾匹配
        if (matchStart >= 0 && matchEnd >= 0) {
            for (int j = matchStart; j <= matchEnd; j++) {
                result.append(replacement);
            }
        }

        return result.toString();
    }

    /**
     * 批量检测敏感词
     * @param text 待检测文本
     * @return 所有匹配到的敏感词列表
     */
    public List<String> findAll(String text) {
        List<String> found = new ArrayList<>();
        if (!initialized || text == null) {
            return found;
        }

        Map<Character, Map<String, Object>> current = dfa;

        for (int i = 0; i < text.length(); i++) {
            char c = Character.toLowerCase(text.charAt(i));

            Map<String, Object> state = current.get(c);

            if (state == null) {
                current = dfa;
                continue;
            }

            if (Boolean.TRUE.equals(state.get("isEnd"))) {
                found.add((String) state.get("word"));
            }

            current = (Map<Character, Map<String, Object>>) state.get("children");
        }

        return found;
    }
}
```

#### 3.5.2 预计送达时间计算

```java
@Service
public class DeliveryETAEngine {

    // 基础配送时间（小时）
    private static final BigDecimal BASE_DELIVERY_HOURS = new BigDecimal("48");

    // 每100公里增加时间（小时）
    private static final BigDecimal HOURS_PER_100KM = new BigDecimal("2");

    // 天气影响系数
    private static final BigDecimal BAD_WEATHER_FACTOR = new BigDecimal("1.3"); // +30%

    @Autowired
    private WeatherService weatherService;

    @Autowired
    private AddressService addressService;

    /**
     * 计算预计送达时间
     *
     * ETA = NOW + 基础时间 × 距离系数 × 天气系数 × 节假日系数
     */
    public LocalDateTime calculateETA(LogisticsInfo logistics, String destination) {
        // 1. 计算基础配送时间
        BigDecimal baseHours = BASE_DELIVERY_HOURS;

        // 2. 计算距离因素
        BigDecimal distanceFactor = calculateDistanceFactor(
            logistics.getOriginCity(),
            destination
        );

        // 3. 天气因素
        BigDecimal weatherFactor = getWeatherFactor(destination);

        // 4. 节假日因素
        BigDecimal holidayFactor = getHolidayFactor();

        // 5. 计算最终时间
        BigDecimal totalHours = baseHours
            .multiply(distanceFactor)
            .multiply(weatherFactor)
            .multiply(holidayFactor)
            .setScale(0, RoundingMode.HALF_UP);

        return LocalDateTime.now().plusHours(totalHours.longValue());
    }

    /**
     * 计算距离系数
     * 距离越远，系数越大
     */
    private BigDecimal calculateDistanceFactor(String origin, String destination) {
        int distance = addressService.calculateDistance(origin, destination);

        // 每增加100公里，系数+0.2
        BigDecimal factor = new BigDecimal(distance)
            .divide(new BigDecimal("100"), 2, RoundingMode.HALF_UP)
            .multiply(HOURS_PER_100KM)
            .divide(new BigDecimal("100"), 2, RoundingMode.HALF_UP);

        // 最低为1.0
        return factor.max(BigDecimal.ONE);
    }

    /**
     * 获取天气影响系数
     */
    private BigDecimal getWeatherFactor(String destination) {
        WeatherInfo weather = weatherService.getWeather(destination);

        if (weather.isSevere()) {
            return BAD_WEATHER_FACTOR; // 恶劣天气 +30%
        } else if (weather.isRainy()) {
            return new BigDecimal("1.15"); // 雨天 +15%
        } else if (weather.isSnowy()) {
            return new BigDecimal("1.2"); // 雪天 +20%
        }

        return BigDecimal.ONE; // 正常天气
    }

    /**
     * 获取节假日系数
     */
    private BigDecimal getHolidayFactor() {
        // 双十一、618等大促期间 +50%
        if (isShoppingFestival(LocalDate.now())) {
            return new BigDecimal("1.5");
        }

        // 春节期间 +30%
        if (isSpringFestivalPeriod(LocalDate.now())) {
            return new BigDecimal("1.3");
        }

        // 节假日 +20%
        if (isHoliday(LocalDate.now())) {
            return new BigDecimal("1.2");
        }

        return BigDecimal.ONE;
    }
}
```

```java
@Service
public class CouponCalculator {

    /**
     * 计算订单最终金额（多优惠券叠加）
     */
    public CouponResult calculate(Order order, List<Coupon> coupons) {
        BigDecimal originalAmount = order.getTotalAmount();
        BigDecimal totalDiscount = BigDecimal.ZERO;

        // 按优先级排序
        coupons.sort((a, b) -> b.getPriority().compareTo(a.getPriority()));

        List<CouponApplyDetail> details = new ArrayList<>();

        BigDecimal remainingAmount = originalAmount;

        for (Coupon coupon : coupons) {
            if (!isApplicable(coupon, order)) {
                continue;
            }

            BigDecimal discount = calculateDiscount(coupon, remainingAmount);

            CouponApplyDetail detail = new CouponApplyDetail();
            detail.setCouponId(coupon.getId());
            detail.setCouponName(coupon.getName());
            detail.setDiscountAmount(discount);
            details.add(detail);

            totalDiscount = totalDiscount.add(discount);
            remainingAmount = remainingAmount.subtract(discount);
        }

        return new CouponResult(originalAmount, totalDiscount, remainingAmount, details);
    }

    /**
     * 计算单张优惠券折扣
     */
    private BigDecimal calculateDiscount(Coupon coupon, BigDecimal amount) {
        BigDecimal discount = BigDecimal.ZERO;

        switch (coupon.getType()) {
            case FIXED:
                // 固定金额（如：减20元）
                discount = coupon.getDiscountValue();
                break;

            case PERCENT:
                // 百分比（如：8折）
                discount = amount
                    .multiply(coupon.getDiscountValue())
                    .setScale(2, RoundingMode.HALF_UP);

                // 最高折扣上限
                if (coupon.getMaxDiscount() != null) {
                    discount = discount.min(coupon.getMaxDiscount());
                }
                break;

            case FREE_SHIPPING:
                // 免邮（特殊处理，不直接减金额）
                discount = order.getFreightAmount();
                break;
        }

        // 折扣不超过订单金额
        return discount.min(amount);
    }
}
```

```java
@Service
public class FlashSaleService {

    @Autowired
    private RedisService redisService;

    /**
     * 库存锁定（Redis原子操作）
     */
    public boolean lockStock(Long productId, Long userId, int quantity) {
        String stockKey = "flash:stock:" + productId;
        String lockKey = "flash:lock:" + productId + ":" + userId;

        // Lua脚本保证原子性
        String luaScript = """
            local stock = tonumber(redis.call('GET', KEYS[1]))
            if stock == nil or stock < tonumber(ARGV[1]) then
                return 0
            end
            redis.call('DECRBY', KEYS[1], ARGV[1])
            return 1
            """;

        Boolean result = redisService.executeLua(luaScript,
            List.of(stockKey),
            String.valueOf(quantity)
        );

        if (result != null && result) {
            // 锁定成功，设置过期时间
            redisService.setex(lockKey, 900, "1"); // 15分钟锁定
            return true;
        }

        return false;
    }

    /**
     * 释放库存（超时未支付）
     */
    public void releaseStock(Long productId, Long userId, int quantity) {
        String stockKey = "flash:stock:" + productId;
        String lockKey = "flash:lock:" + productId + ":" + userId;

        // 释放锁定库存
        redisService.incrby(stockKey, quantity);
        redisService.del(lockKey);
    }
}
```

```java
@Service
public class JwtTokenProvider {

    public String generateToken(UserDetails user) {
        Date now = new Date();
        Date expiry = new Date(now.getTime() + 86400000); // 24小时

        return Jwts.builder()
            .setSubject(user.getUsername())
            .claim("role", user.getRole())
            .setIssuedAt(now)
            .setExpiration(expiry)
            .signWith(key, SignatureAlgorithm.HS512)
            .compact();
    }

    public boolean validateToken(String token) {
        try {
            Jwts.parserBuilder()
                .setSigningKey(key)
                .build()
                .parseClaimsJws(token);
            return true;
        } catch (JwtException e) {
            return false;
        }
    }
}
```

```java
@Service
public class LotteryService {

    @Transactional
    public DrawResult draw(Long userId, Long activityId) {
        // 1. 检查活动是否进行中
        LotteryActivity activity = activityRepository.findById(activityId)
            .orElseThrow(() -> new BusinessException("活动不存在"));

        if (!activity.isActive()) {
            throw new BusinessException("活动未开始或已结束");
        }

        // 2. 检查用户积分是否足够
        User user = userRepository.findById(userId)
            .orElseThrow(() -> new BusinessException("用户不存在"));

        int cost = activity.getCreditCost();
        if (user.getCredit() < cost) {
            throw new BusinessException("积分不足");
        }

        // 3. 尝试获取分布式锁
        String lockKey = "lottery:lock:" + userId + ":" + activityId;
        if (!redisLock.tryLock(lockKey, 30, TimeUnit.SECONDS)) {
            throw new BusinessException("操作太频繁");
        }

        try {
            // 4. 扣减积分
            creditService.deductCredit(userId, cost, "LOTTERY", "抽奖消耗");

            // 5. 执行抽奖
            LotteryPrize prize = doDraw(activity);

            // 6. 记录
            if (prize != null) {
                LotteryRecord record = new LotteryRecord();
                record.setUser(user);
                record.setActivity(activity);
                record.setPrize(prize);
                record.setStatus("WON");
                lotteryRecordRepository.save(record);

                return new DrawResult(true, prize);
            } else {
                return new DrawResult(false, null);
            }

        } finally {
            redisLock.unlock(lockKey);
        }
    }

    private LotteryPrize doDraw(LotteryActivity activity) {
        List<LotteryPrize> prizes = activity.getPrizes();

        // 计算概率
        BigDecimal random = BigDecimal.valueOf(Math.random());
        BigDecimal cumulative = BigDecimal.ZERO;

        for (LotteryPrize prize : prizes) {
            BigDecimal prob = adjustProbability(prize);
            cumulative = cumulative.add(prob);

            if (random.compareTo(cumulative) <= 0) {
                // 中奖
                prizeRepository.decrementStock(prize.getId());
                return prize;
            }
        }

        return null; // 未中奖
    }
}
```

```java
@Service
public class CouponService {

    @Transactional
    public void receiveCoupon(Long userId, Long templateId) {
        CouponTemplate template = templateRepository.findById(templateId)
            .orElseThrow(() -> new BusinessException("优惠券不存在"));

        // 检查库存
        if (template.getRemainingCount() <= 0) {
            throw new BusinessException("优惠券已领完");
        }

        // 检查是否已领取
        if (userCouponRepository.existsByUserIdAndTemplateId(userId, templateId)) {
            throw new BusinessException("您已领取过该优惠券");
        }

        // 创建用户优惠券
        UserCoupon userCoupon = new UserCoupon();
        userCoupon.setUser(userRepository.getReferenceById(userId));
        userCoupon.setCouponTemplate(template);
        userCoupon.setStatus("UNUSED");
        userCoupon.setReceivedAt(LocalDateTime.now());
        userCoupon.setValidFrom(template.getValidFrom());
        userCoupon.setValidTo(template.getValidTo());

        userCouponRepository.save(userCoupon);

        // 扣减库存
        templateRepository.decrementRemainingCount(templateId);
    }

    @Transactional
    public BigDecimal calculateDiscount(Long userId, Long couponId,
            BigDecimal orderAmount) {
        UserCoupon userCoupon = userCouponRepository.findById(couponId)
            .orElseThrow(() -> new BusinessException("优惠券不存在"));

        CouponTemplate template = userCoupon.getCouponTemplate();

        // 检查是否满足使用条件
        if (orderAmount.compareTo(template.getMinAmount()) < 0) {
            throw new BusinessException(
                String.format("订单金额需满%.2f元", template.getMinAmount()));
        }

        // 计算折扣
        BigDecimal discount;
        if ("FIXED".equals(template.getDiscountType())) {
            discount = template.getDiscountValue();
        } else {
            discount = orderAmount.multiply(template.getDiscountValue());
            if (template.getMaxDiscount() != null) {
                discount = discount.min(template.getMaxDiscount());
            }
        }

        return discount.min(orderAmount);
    }
}
```
